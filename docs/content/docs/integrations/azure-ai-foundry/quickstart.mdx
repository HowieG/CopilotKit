## Prerequisites

Before you begin, you'll need the following:

- An [Azure AI Foundry](https://ai.azure.com) project with a deployed model (e.g. GPT-4o)
- Azure CLI installed and logged in (`az login`)
- Python 3.10+ and [uv](https://docs.astral.sh/uv/) (for the agent backend)
- Node.js 20+
- Your favorite package manager (npm, pnpm, yarn, or bun)

                ### Install dependencies

                The starter includes a `postinstall` script that automatically installs both your npm and agent dependencies.


                <Callout type="info" title="Manual setup">
                    If you have issues with automatic Python setup, you can manually install the agent dependencies:
                    ```bash

                    uv sync
                    ```

                ### Configure your environment

                Create a `.env` file inside the `agent` folder with your Azure AI Foundry configuration:

                ```bash title="agent/.env"
                AZURE_AI_FOUNDRY_PROJECT_ENDPOINT=https://your-project.services.ai.azure.com
                AZURE_OPENAI_DEPLOYMENT_NAME=gpt-4o
                ```

                <Callout type="info" title="Authentication">
                    The starter uses `DefaultAzureCredential`, which supports:
                    - **Azure CLI** — run `az login` before starting (recommended for development)
                    - **Managed Identity** — works automatically when deployed to Azure
                    - **API Key** — set `AZURE_AI_FOUNDRY_API_KEY` in your `.env` if you prefer key-based auth

                    See [Azure Identity documentation](https://learn.microsoft.com/en-us/python/api/overview/azure/identity-readme) for all supported credential types.
                </Callout>

            <Step>
                ### Set up your Azure AI Foundry agent

                Create a FastAPI server that connects to Azure AI Foundry and exposes an AG-UI endpoint:

                ```python title="agent/src/agent.py"
                from __future__ import annotations

                import os

                import uvicorn
                from azure.identity import DefaultAzureCredential
                from azure.ai.projects import AIProjectClient
                from agent_framework.azure import AzureOpenAIChatClient
                from agent_framework import ChatAgent
                from agent_framework.ag_ui import add_agent_framework_fastapi_endpoint
                from dotenv import load_dotenv
                from fastapi import FastAPI


                load_dotenv()

                # Connect to Azure AI Foundry
                project_endpoint = os.getenv("AZURE_AI_FOUNDRY_PROJECT_ENDPOINT")
                deployment_name = os.getenv("AZURE_OPENAI_DEPLOYMENT_NAME", "gpt-4o")

                credential = DefaultAzureCredential()

                # Create an Azure OpenAI chat client via the Foundry project
                chat_client = AzureOpenAIChatClient(
                    credential=credential,
                    deployment_name=deployment_name,
                    endpoint=project_endpoint,
                )

                agent = ChatAgent(
                    name="FoundryAgent",
                    instructions="You are a helpful assistant powered by Azure AI Foundry.",
                    chat_client=chat_client,
                )

                app = FastAPI(title="Azure AI Foundry Agent")
                add_agent_framework_fastapi_endpoint(app=app, agent=agent, path="/")


                if __name__ == "__main__":
                    uvicorn.run("agent:app", host="0.0.0.0", port=8000, reload=True)
                ```

                Set your environment variables and run:

                ```bash title="agent/.env"
                AZURE_AI_FOUNDRY_PROJECT_ENDPOINT=https://your-project.services.ai.azure.com
                AZURE_OPENAI_DEPLOYMENT_NAME=gpt-4o
                ```

                ```bash
                cd agent
                uv run src/agent.py
                ```

                ```npm

                ```

            <Step>
                ### Setup Copilot Runtime

                CopilotKit requires a Copilot Runtime endpoint to communicate with your agent.

                Create a new API route at `app/api/copilotkit/route.ts`:

                ```tsx title="app/api/copilotkit/route.ts"
                import {
                  CopilotRuntime,
                  ExperimentalEmptyAdapter,
                  copilotRuntimeNextJSAppRouterEndpoint,
                } from "@copilotkit/runtime";
                import { HttpAgent } from "@ag-ui/client";
                import { NextRequest } from "next/server";

                const serviceAdapter = new ExperimentalEmptyAdapter();

                // [!code highlight:5]
                const runtime = new CopilotRuntime({
                  agents: {
                    foundry_agent: new HttpAgent({ url: "http://localhost:8000/" }),
                  },
                });

                export const POST = async (req: NextRequest) => {
                  const { handleRequest } = copilotRuntimeNextJSAppRouterEndpoint({
                    runtime,
                    serviceAdapter,
                    endpoint: "/api/copilotkit",
                  });

                  return handleRequest(req);
                };
                ```
            </Step>
            <Step>
                ### Configure CopilotKit Provider

                Wrap your application with the CopilotKit provider:

                ```tsx title="app/layout.tsx"
                import { CopilotKit } from "@copilotkit/react-core/v2"; // [!code highlight]
                import "@copilotkit/react-ui/v2/styles.css";

                export default function RootLayout({ children }: {children: React.ReactNode}) {
                  return (
                    <html lang="en">
                      <body>
                        {/* [!code highlight:3] */}
                        <CopilotKit runtimeUrl="/api/copilotkit" agent="foundry_agent">
                          {children}
                        </CopilotKit>
                      </body>
                    </html>
                  );
                }
                ```
            </Step>
            <Step>

              ```tsx title="app/page.tsx"
              export default function Page() {
              ```

            <Accordion title="Troubleshooting">
                **Agent Connection Issues**
                - If you see "I'm having trouble connecting to my tools", make sure:
                  - The Python agent is running on port 8000
                  - Your Azure credentials are configured correctly (`az login`)
                  - Both servers started successfully (check terminal output)

                **Azure Authentication Issues**
                - If the agent fails with credential errors:
                  ```bash
                  # Make sure you're logged in
                  az login
                  # Verify your subscription
                  az account show
                  ```

                **Python Environment Issues**
                - Verify Python and uv are installed:
                  ```bash
                  python --version  # Should be 3.10+
                  uv --version
                  ```
                - Reinstall dependencies:
                  ```bash
                  cd agent
                  uv sync
                  ```

                **Port Conflicts**
                - If port 8000 is already in use, update it in:
                  - `agent/src/agent.py` — change the `uvicorn.run` port
                  - `src/app/api/copilotkit/route.ts` — update the HttpAgent URL
            </Accordion>
